include(${CMAKE_CURRENT_LIST_DIR}/../cmake/MNN.cmake)

add_library(sd15_runtime STATIC src/runtime.cpp src/diffusion.cpp)
target_include_directories(sd15_runtime PUBLIC include)
target_link_libraries(sd15_runtime PRIVATE sd15_mnn)
target_compile_features(sd15_runtime PUBLIC cxx_std_17)
if(MSVC)
    target_compile_options(sd15_runtime PRIVATE /W4 /utf-8 /fp:strict)
    target_compile_definitions(sd15_runtime PRIVATE NOMINMAX)
else()
    target_compile_options(sd15_runtime PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()

if(BUILD_TESTING)
    add_executable(sd15-runtime-test tests/diffusion_test.cpp src/diffusion.cpp)
    target_include_directories(sd15-runtime-test PRIVATE include src)
    target_compile_features(sd15-runtime-test PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(sd15-runtime-test PRIVATE /utf-8 /fp:strict)
    endif()
    add_test(NAME runtime_math COMMAND sd15-runtime-test)
endif()
