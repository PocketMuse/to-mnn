include(${CMAKE_CURRENT_LIST_DIR}/../cmake/MNN.cmake)
find_package(nlohmann_json 3.11 CONFIG REQUIRED)

find_package(ICU REQUIRED COMPONENTS uc i18n)

add_library(sd15_runtime STATIC src/runtime.cpp src/diffusion.cpp src/clip_tokenizer.cpp
    src/mnn_model_session.cpp src/sd15_models.cpp)
target_include_directories(sd15_runtime PUBLIC include)
target_link_libraries(sd15_runtime PRIVATE sd15_mnn ICU::uc ICU::i18n nlohmann_json::nlohmann_json)
target_compile_features(sd15_runtime PUBLIC cxx_std_17)
# 준비된 엔진의 빌드 설정을 확인한다. 옵션이 조용히 무시되는 것을 방지한다.
if(EXISTS "${MNN_RUNTIME_DIR}/CMakeCache.txt")
    file(STRINGS "${MNN_RUNTIME_DIR}/CMakeCache.txt" mnn_low_memory REGEX "^MNN_LOW_MEMORY:BOOL=(ON|TRUE|1)$")
    if(mnn_low_memory)
        target_compile_definitions(sd15_runtime PRIVATE SD15_MNN_LOW_MEMORY)
    endif()
endif()
if(MSVC)
    target_compile_options(sd15_runtime PRIVATE /W4 /utf-8 /fp:strict)
    target_compile_definitions(sd15_runtime PRIVATE NOMINMAX)
else()
    target_compile_options(sd15_runtime PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()

if(BUILD_TESTING)
    add_executable(sd15-tokenizer-test tests/tokenizer_test.cpp)
    target_include_directories(sd15-tokenizer-test PRIVATE src)
    target_link_libraries(sd15-tokenizer-test PRIVATE sd15_runtime Catch2::Catch2)
    if(MSVC)
        target_compile_options(sd15-tokenizer-test PRIVATE /utf-8)
        add_custom_command(TARGET sd15-tokenizer-test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "$<TARGET_FILE:sd15_mnn>" "$<TARGET_FILE_DIR:sd15-tokenizer-test>")
    endif()
    set(SD15_TEST_TOKENIZER_DIR "" CACHE PATH "Optional real CLIP tokenizer directory for tests")
    if(SD15_TEST_TOKENIZER_DIR)
        add_test(NAME runtime_tokenizer COMMAND sd15-tokenizer-test --tokenizer-dir "${SD15_TEST_TOKENIZER_DIR}")
    endif()
    add_executable(sd15-generation-test tests/generation_test.cpp)
    target_include_directories(sd15-generation-test PRIVATE src)
    target_link_libraries(sd15-generation-test PRIVATE sd15_runtime sd15_mnn Catch2::Catch2)
    if(MSVC)
        target_compile_options(sd15-generation-test PRIVATE /utf-8)
        add_custom_command(TARGET sd15-generation-test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "$<TARGET_FILE:sd15_mnn>" "$<TARGET_FILE_DIR:sd15-generation-test>")
    endif()
    set(SD15_TEST_MODEL_DIR "" CACHE PATH "Optional real SD1.5 model directory for OpenCL tests")
    if(SD15_TEST_MODEL_DIR AND SD15_TEST_TOKENIZER_DIR)
        add_test(NAME runtime_generation COMMAND sd15-generation-test --model-dir "${SD15_TEST_MODEL_DIR}" --tokenizer-dir "${SD15_TEST_TOKENIZER_DIR}")
        set_tests_properties(runtime_generation PROPERTIES TIMEOUT 600)
    endif()
    add_executable(sd15-runtime-test tests/diffusion_test.cpp src/diffusion.cpp)
    target_include_directories(sd15-runtime-test PRIVATE include src)
    target_compile_features(sd15-runtime-test PRIVATE cxx_std_17)
    target_link_libraries(sd15-runtime-test PRIVATE Catch2::Catch2WithMain)
    if(MSVC)
        target_compile_options(sd15-runtime-test PRIVATE /utf-8 /fp:strict)
    endif()
    catch_discover_tests(sd15-runtime-test TEST_PREFIX "runtime_math/")
endif()
